# ── Eyes ────────────────────────────────────────────────────────────────────

# region so they stay distinct while still blending toward the middle.

                                    "mirror"                · "alternating" · Eyes
                       #   uniform     = both eyes spin the same way
                       #   alternating = eyes spin in opposite directions
                       #   mirror      = opposite spin, but both pull in/out together

eyeSoftness            · number   · 0 – 100 (%)             · 60          · Eyes (edge falloff softness)
